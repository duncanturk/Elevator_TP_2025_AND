\begin{frame}{Invariants for Requirements: Elevator Movement}
	\begin{itemize}
		\item<+-> \Req{FUN 16}{The elevator should not move in a direction where there are no pending requests to attend.}
		\item[]<+-> \Inv{inv\_move\_to\_ordered}{destination$\neq$NONE$\implies$destination$\in$orders}
		\item<+-> \Req{FUN 15}{The elevator should not move if there are no requests waiting to be served.}

\subsection{Informal Argument for the Liveness of the Algorithm} % Antonio

\begin{frame}{Informal Argument for the Liveness of the Algorithm}
	\begin{itemize}
		\item<+-> Properties this algorithm has:
		\begin{itemize}
			\item<+-> The elevator moves/starts moving when there are any orders.
			\item<+-> The elevator stops at every floor that has been ordered and that is in its immediate path.
			\item<+-> If the elevator is going up/down, then it continues to do so as long as there are orders above/below.
			\item<+-> When it runs out of orders above/below, but it still has orders the other way, it changes direction.
		\end{itemize}
		\item<+-> Then, for any floor that is ordered:
		\begin{itemize}
			\item<+-> The elevator may be at that floor already, so it is served. % TODO revise
			\item<+-> The elevator is not currently at that floor. Then either:
			\begin{itemize}
				\item<+-> It's moving in the opposite direction, in which case it will eventually turn around.
				\item<+-> It's moving in the correct direction, in which case it will eventually reach the ordered floor and serve it.
			\end{itemize}
		\end{itemize}
	\end{itemize}
\end{frame}

\subsection{Deadlock Freedom} % Antonio

\begin{frame}{Deadlock Freedom}

	\end{itemize}

\subsection{State Transitions} % Antonio

\begin{frame}{State Transitions}
	\begin{itemize}

	\end{itemize}

\subsection{Possible Improvements} % Christopher

\subsubsection{Door Modelling} % Christopher

\begin{frame}{Possible Improvements: Door Modelling} % Christopher
	\begin{itemize}
		\item The modelling of the physical door state is not completely "clean"
		\begin{itemize}
			\item E.g. the \Constant{door} can be \Constant{CLOSED} while the door is currently opening
		\end{itemize}
		\item Options to fix this came to our minds:
		\begin{enumerate}
			\item Add additional physical states \Constant{OPENING} and \Constant{CLOSING}
			\begin{itemize}
				\item New environment events \Event{EnvStartOpening} and \Event{StartClosing} would be needed to adapt the physical state when \Guard{door$\neq$desired\_door}
				\item This adds complexity to the model
				\item It contradicts the idea, that \Variable{desired\_door=OPEN} instantly triggers the opening
			\end{itemize}
			\item Add a function \Variable{physical\_door\_state$\in$(DOOR$\times$DOOR)$\to$\{OPEN, OPENING, CLOSED, CLOSING\}}, that interprets \Variable{door} and \Variable{desired\_door}
			\begin{itemize}
				\item Usage would change for example the invariant \InvariantName{inv\_no\_orders\_door\_open} which was
				\Invariant{orders=$\emptyset\implies$door=OPEN\,$\lor$\,desired\_door=OPEN}
				\\to
				\Invariant{orders=$\emptyset\implies$(physical\_door\_state(door, desired\_door)$\in$\{OPEN, OPENING\})}
				\item This can improve readability
				\item But in increases proof complexity
			\end{itemize}
			\item Use a single variable for \Variable{door} which is one of \Constant{OPEN, OPENING, CLOSED, CLOSING}
			\begin{itemize}
				\item Setting \Action{door:=CLOSING} by the controller would make the door start closing
				\item The environment would set this variable upon finishing to \Action{door:=CLOSED}
				\item This contradicts the idea, that the variables are only changed by the either environment or the controller
				\item This exception may be acceptable since \Variable{orders} behaves in a similar way
			\end{itemize}
		\end{enumerate}
	\end{itemize}
\end{frame}

\subsubsection{Reduce Direction Events} % Christopher

\begin{frame}{Reduce Direction Events} % Christopher
	\begin{itemize}
		\item We can define a helper function \Constant{cut(floor, orders, direction)} that returns the remaining orders in the direction of the current movement
		\item This functions has the axioms:
		\begin{itemize}
			\item \Inv{axm\_cut\_type}{cut$\in$(FLOOR$\times \mathbb{P}$(FLOOR)$\times$DIRECTION)$\implies \mathbb{P}$(FLOOR)}
			\item \Inv{axm\_cut\_up}{\\
				$\forall$floor,orders,direction$\cdot$\\
			\hfill floor$\in$FLOOR$\land$orders$\subseteq$FLOOR$\land$direction$\in$DIRECTION$\land$direction=UP\\
			\hfill$\implies$cut(floor$\mapsto$orders$\mapsto$direction)=\{x$\vert$ x$\in$orders$\land$x>floor\}}
			\item \Inv{axm\_cut\_down}{\\
				$\forall$floor,orders,direction$\cdot$\\
				\hfill floor$\in$FLOOR$\land$orders$\subseteq$FLOOR$\land$direction$\in$DIRECTION$\land$direction=DOWN\\
				\hfill$\implies$cut(floor$\mapsto$orders$\mapsto$direction)=\{x$\vert$ x$\in$orders$\land$x<floor\}}
		\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Reduce Direction Events}
	\begin{itemize}
		\item This would lead to reduction in events and avoids copy-paste-errors:
		\begin{itemize}
			\item events \Event{ChangeDirectionUp} and \Event{ChangeDirectionDown} can be unified
			\begin{itemize}
				\item Common guards and actions stay
				\item The guard \Guard{cut(floor$\mapsto$orders$\mapsto$direction)=$\emptyset$}
				\item and the action \Action{direction:$\in$DIRECTION$\setminus$\{direction\}} are added
			\end{itemize}
			\item events \Event{ChooseNextUp} and \Event{ChooseNextDown} can be unified
			\begin{itemize}
				\item Common guards and actions stay
				\item The guard \Guard{cut(floor$\mapsto$orders$\mapsto$direction)$\neq \emptyset$}
				\item and the action \Action{direction:$\in$cut(floor$\mapsto$orders$\mapsto$direction)}
				\item To choose deterministic \Constant{max} or \Constant{min} could be used but this is not necessary
				\item Even a function like \Constant{next\_cut} that chooses the \Constant{max} or \Constant{min} of the \Constant{cut}-set depending on the \Constant{direction} is possible
			\end{itemize}
		\end{itemize}
		\item We implemented this but a lot of proofs required repeated manual intervention as we made changes to the model
		\item Using the separate events made the proof obligations discharge automatically which was a big help in the iterative development, so we chose the separate event approach
	\end{itemize}
\end{frame}
