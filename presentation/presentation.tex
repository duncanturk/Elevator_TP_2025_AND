\subsection{Elevator Movement}
\begin{frame}{Elevator Movement}
	\begin{itemize}
		\item<+-> Physical movement modeled through \Event{EnvMove}
			\begin{itemize}
				\item<+-> Moving one step at a time \Guard{(new\_floor-floor) $\in\{-1,1\}$}
				\item<+-> In the direction of the destination \Guard{new\_floor$\in$floor..destination}
				\item<+-> So the \Variable{next\_floor} is one above \Variable{floor} if \Variable{destination} is above \Variable{floor} and one below otherwise
			\end{itemize}
		\item<+-> Handling of movement events by \Event{ReachOrder} or \Event{ReachNotOrder}
		\item<+-> Ensuring the handling of every movement event through \Variable{floor\_event} $\in$ \Constant{\{PENDING, HANDLED\}} flag
		\item<+-> TODO graphic of movement \Event{$test\in$}
	\end{itemize}
\end{frame}
\begin{frame}{Elevator Movement}
	\begin{itemize}
		\item<+-> \Req{FUN 16}{The elevator should not move in a direction where there are no pending requests to attend.}
		\item[]<+-> \Inv{inv\_move\_to\_ordered}{destination$\neq$NONE$\implies$destination$\in$orders}
		\item<+-> \Req{FUN 15}{The elevator should not move if there are no requests waiting to be served.}
		\item[]<+-> \Inv{inv\_only\_move\_with\_orderes}{movement=MOVING$\implies$orders$\neq\emptyset$}
		\item[]<.-> This follows directly from \Requirement{FUN 16} and the helper: \Inv{inv\_only\_move\_when\_destination}{movement=MOVING$\implies$destination$\neq$NONE}
	\end{itemize}
\end{frame}
\subsection{Door Management}
\begin{frame}{Door Management}
	\begin{itemize}
		\item<+-> Necessary to ensure movement restrictions according to door status
		\item[]<+-> \begin{tabularx}{\linewidth}{|l|c|X|}
					\hline
					\Variable{door} & \Variable{desired\_door} & Meaning \\ \hline
					\Constant{OPEN} & \Constant{OPEN} & Door is open and stays open \\ \hline
					\Constant{OPEN} & \Constant{CLOSED} & The door is closing or may be closed by the environment asap \\ \hline
					\Constant{CLOSED} & \Constant{OPEN} & The door is opening or may be opened by the environment asap \\ \hline
					\Constant{CLOSED} & \Constant{CLOSED} & The door is closed and stays closed \\ \hline
				\end{tabularx}
		\item \Req{FUN 8}{The elevator doors should be closed when the elevator is moving.}
		\item[]<+-> \Inv{inv\_door\_closed\_when\_moving}{\\movement=MOVING$\implies$door=CLOSED$\land$desired\_door=CLOSED}
		\item[]<+-> Helper:
		\item[]<.-> \Inv{inv\_door\_closed\_when\_destination}{destination$\neq$NONE$\implies$door=CLOSED}
		\item[]<.-> \Inv{inv\_no\_dest\_when\_opening\_door}{desired\_door=OPEN$\implies$destination=NONE}
		\item<+-> \Req{FUN 11}{The doors should be open if there are no requests to serve.}
		\item[]<+-> \Inv{inv\_no\_orders\_door\_open}{orders=$\emptyset\implies$(door=OPEN\,$\lor$\,desired\_door=OPEN)}
		\item[]<+-> Helper:
		\item[]<.-> \Inv{inv\_orders\_when\_closing\_door}{desired\_door=CLOSED$\implies$orders$\neq\emptyset$}
	\end{itemize}
\end{frame}
\subsection{Refinement: Deterministic Serving Order}
\begin{frame}{Refinement: Deterministic Serving Order}
	\begin{itemize}
		\item<+-> So far: \Event{ChooseNext} selects \Action{destination:$\in$orders} randomly
		\item[]<+-> \Req{FUN 13}{All pending requests should be eventually be served.}
		\item<+-> Imagine this situation:
		\begin{tabularx}{\linewidth}{|c|c|X|}\hline
			\Variable{floor} & \Variable{orders} & Description \\\hline
			\Variable{0} & \Variable{\{1,4\}} & We are at the ground floor.
			Floors 1 and 4 are ordered. \\\hline
			\Variable{1} & \Variable{\{4\}} & We select floor 1 and served it. \\\hline
			\Variable{1} & \Variable{\{0,4\}} & Somebody ordered the ground floor. \\\hline
			\Variable{0} & \Variable{\{4\}} & We select floor 0 and served it. \\\hline
			\Variable{0} & \Variable{\{1,4\}} & Somebody ordered floor 1.
			This can repeat infinitely.\\\hline
		\end{tabularx}
	\end{itemize}
\end{frame}
